const { onDocumentCreated } = require("firebase-functions/v2/firestore"); const { defineSecret } = require("firebase-functions/params"); const { initializeApp } = require("firebase-admin/app"); const { getStorage } = require("firebase-admin/storage"); const { SolapiMessageService } = require("solapi"); initializeApp(); // Firebase 콘솔이 아니라 터미널(또는 Cloud Console)에서 아래 3개 비밀값을 등록해야 합니다. // firebase functions:secrets:set SOLAPI_API_KEY // firebase functions:secrets:set SOLAPI_API_SECRET // firebase functions:secrets:set SENDER_PHONE (예: 0212345678 — 솔라피에 등록한 학원 발신번호, 숫자만) const SOLAPI_API_KEY = defineSecret("SOLAPI_API_KEY"); const SOLAPI_API_SECRET = defineSecret("SOLAPI_API_SECRET"); const SENDER_PHONE = defineSecret("SENDER_PHONE"); exports.sendEnrollmentSms = onDocumentCreated( { document: "enrollments/{docId}", region: "asia-northeast3", // Firestore를 서울 리전으로 만들었다면 맞춰주세요. 다르면 이 값을 바꿔주세요. secrets: [SOLAPI_API_KEY, SOLAPI_API_SECRET, SENDER_PHONE], }, async (event) => { const snap = event.data; if (!snap) return; const docId = event.params.docId; let data = snap.data(); // 등록 원서 저장 → PDF 생성 → Storage 업로드 → 같은 문서에 pdfPath 업데이트, // 이 순서가 앱에서 순차적으로 일어나기 때문에, onCreate 시점엔 pdfPath가 // 아직 없을 수 있습니다. 최대 10초 정도 짧게 재확인합니다. let pdfPath = data.pdfPath; let tries = 0; while (!pdfPath && tries < 10) { await new Promise((r) => setTimeout(r, 1000)); const fresh = await snap.ref.get(); data = fresh.data(); pdfPath = data.pdfPath; tries++; } if (!pdfPath) { console.error("PDF 경로를 찾지 못해 문자 발송을 건너뜁니다:", docId); return; } const bucket = getStorage().bucket(); const file = bucket.file(pdfPath); const [url] = await file.getSignedUrl({ action: "read", expires: Date.now() + 1000 * 60 * 60 * 24 * 30, // 30일간 유효한 링크 }); const messageService = new SolapiMessageService( SOLAPI_API_KEY.value(), SOLAPI_API_SECRET.value() ); const studentName = data.name || "학생"; const text = `[주니퍼 실용음악학원]\n` + `${studentName} 학생의 등록 원서가 접수되었습니다.\n` + `아래 링크에서 원서를 확인해 주세요.\n${url}`; const recipients = [data.studentPhone, data.parentPhone] .filter(Boolean) .map((p) => p.replace(/\D/g, "")); for (const to of recipients) { try { await messageService.send({ to, from: SENDER_PHONE.value(), text, }); console.log("문자 발송 성공:", to); } catch (e) { console.error("문자 발송 실패:", to, e); } } } );
